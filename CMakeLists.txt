# One static library per engine/library module, then the game on top.
function(jet_module name dir)
  file(GLOB_RECURSE src CONFIGURE_DEPENDS ${dir}/*.c ${dir}/*.cc ${dir}/*.cpp)
  # the editor (source/editor, which has a main()) has its own targets below, not part of the game library
  list(FILTER src EXCLUDE REGEX "/${dir}/editor/")  # GLOB returns full paths
  if(src)
    add_library(${name} STATIC ${src})
  else()
    add_library(${name} INTERFACE)
  endif()
endfunction()

# <mpc/x.h> and "x.h" (same-dir in the original libmpcdec layout), so both roots must be on the path

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")

endif()

option(JET_TESTS "Build core tests" OFF)
if(JET_TESTS)
  enable_testing()
  file(GLOB tests CONFIGURE_DEPENDS tests/test_*.cpp)
  # the game data folder (unzip data.zip there); the tests that need it are registered only when it is present
  set(JET_DATA_DIR "${CMAKE_SOURCE_DIR}/data" CACHE PATH "game data folder for the data tests")
  set(JET_DATA_TESTS test_babel test_pig_roundtrip test_project_import test_intro)
  # tools print or convert and need arguments: built, not run by ctest (see README)
  set(JET_TOOLS test_mpc test_stream_dump test_states test_intro_dump)
  foreach(t ${tests})
    get_filename_component(n ${t} NAME_WE)
    add_executable(${n} ${t})
    target_link_libraries(${n} game core_jet core_claraloader core_net lib_vox1_1)
    target_include_directories(${n} PRIVATE source tests)
    if(n IN_LIST JET_TOOLS)
      # built only
    elseif(n IN_LIST JET_DATA_TESTS)
      if(EXISTS "${JET_DATA_DIR}/datalibs")
        if(n STREQUAL "test_pig_roundtrip")
          set(args "${JET_DATA_DIR}/models")
        elseif(n STREQUAL "test_project_import")
          set(args "${JET_DATA_DIR}" "${CMAKE_BINARY_DIR}/project_import_work")
        elseif(n STREQUAL "test_intro")
          set(args "${JET_DATA_DIR}" "${CMAKE_BINARY_DIR}/test_intro" 480 854 4000)
        else()
          set(args "${JET_DATA_DIR}")
        endif()
        add_test(NAME ${n} COMMAND ${n} ${args})
      endif()
    else()
      add_test(NAME ${n} COMMAND ${n})
    endif()
  endforeach()
  target_link_libraries(test_pig_roundtrip editor_formats)
  target_link_libraries(test_project_import editor_project)
  if(APPLE)
    file(GLOB gui_tests CONFIGURE_DEPENDS tests/test_*.mm)
    foreach(t ${gui_tests})
      get_filename_component(n ${t} NAME_WE)
      add_executable(${n} ${t})
      target_link_libraries(${n} game core_jet core_claraloader core_net "-framework Cocoa")
      target_include_directories(${n} PRIVATE source tests)
    endforeach()
  endif()
endif()
